settingsModal.classList.remove("open"); showMessage("Настройки НЕЙРЫ сохранены."); }; // Состояния ядра function setState(state, label) { orb.className = "orb " + state; stateLabel.textContent = label; } // Сообщения function showMessage(message, error = false) { chatBox.classList.toggle("error", error); chatBox.textContent = message; chatBox.scrollTop = chatBox.scrollHeight; } // Голосовой ответ function speak(message) { if (!window.speechSynthesis) return; try { const currentContents = [ ...history, { role: "user", parts } ]; const response = await fetch(API_URL, { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ model: config.model, systemInstruction: { parts: [{ text: config.prompt }] }, contents: currentContents, generationConfig: { temperature: 0.7, maxOutputTokens: 4096 } }) }); const data = await response.json(); if (!response.ok) { throw new Error( data.error?.message || "Ошибка соединения с сервером." ); } const answer = data.candidates?.[0]?.content?.parts ?.filter(p => p.text) .map(p => p.text) .join("\n"); if (!answer) { throw new Error("НЕЙРА не получила текстовый ответ."); } history.push( { role: "user", parts }, { role: "model", parts: [{ text: answer }] } ); // Храним последние 10 пар сообщений if (history.length > 20) { history = history.slice(-20); } showMessage(answer); setState("", "ОЖИДАНИЕ"); speak(answer); selectedImage = null; selectedMime = null; fileInput.value = ""; previewContainer.classList.remove("active"); } catch (error) { showMessage("ОШИБКА НЕЙРЫ: " + error.message, true); setState("", "ОШИБКА"); console.error(error); } finally { busy = false; } } // Отправка текста function sendMessage() { const message = textInput.value; if (!message.trim() && !selectedImage) return; if (!busy) setState("", "ОЖИДАНИЕ"); }; recognition.onerror = () => { micBtn.classList.remove("active"); setState("", "ОШИБКА МИКРОФОНА"); }; } else { micBtn.onclick = () => { alert("Голосовой ввод не поддерживается."); }; } showMessage("НЕЙРА активирована.
